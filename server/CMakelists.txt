cmake_minimum_required(VERSION 3.16)
project(UDS_OB_HFT CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_BUILD_TYPE Release)
add_compile_options(-O2 -Wall -Wextra)

add_executable(server
    main.cpp
    market_data_feed.cpp
    session_manager.cpp
)
target_include_directories(server PRIVATE ../common)
target_link_libraries(server pthread)

add_executable(data_reader_sender
    data_reader_sender.cpp
)
target_include_directories(data_reader_sender PRIVATE ../common)
